import os, json, time, requests, urllib3, markdown
from flask import Flask, render_template_string, request
from google import genai

urllib3.disable_warnings()  # Proxmox uses a self-signed cert

# ---------- Settings ----------
PVE_HOST = os.environ.get("PVE_HOST", "192.168.67.2")
PVE_TOKEN = os.environ["PVE_TOKEN"]          # report@pam!reporting=SECRET
MODEL = "gemini-3.6-flash"
PORT = 8000
# ------------------------------

client = genai.Client()  # reads GEMINI_API_KEY from the environment
app = Flask(__name__)

KEEP = ["id", "type", "name", "node", "status", "cpu", "maxcpu", "mem",
        "maxmem", "disk", "maxdisk", "uptime", "netin", "netout"]


def get_proxmox_data():
    r = requests.get(
        f"https://{PVE_HOST}:8006/api2/json/cluster/resources",
        headers={"Authorization": f"PVEAPIToken={PVE_TOKEN}"},
        verify=False,
        timeout=15,
    )
    r.raise_for_status()
    return [{k: x[k] for k in KEEP if k in x} for x in r.json()["data"]]


def make_report():
    """Returns (report_markdown, error_message)."""
    try:
        data = get_proxmox_data()
    except Exception as e:
        return None, f"Couldn't get data from Proxmox: {e}"

    prompt = f"""Write a concise report on this Proxmox homelab.
Convert bytes to GB, CPU to percent, and uptime to days.
Cover: overall health, resource usage per VM/container,
storage capacity, and recommendations.
Base conclusions only on this data.

{json.dumps(data)}"""

    for attempt in range(3):
        try:
            response = client.models.generate_content(model=MODEL, contents=prompt)
            return response.text, None
        except Exception as e:
            msg = str(e)
            if "503" in msg or "UNAVAILABLE" in msg:
                if attempt < 2:
                    time.sleep(5)
                    continue
                return None, "The AI service is busy right now. Please try again later."
            return None, f"Something went wrong: {msg}"

@app.route("/", methods=["GET", "POST"])
def index():
    report_html, error = None, None
    if request.method == "POST":
        text, error = make_report()
        if text:
            report_html = markdown.markdown(text, extensions=["tables"])
    return render_template_string(PAGE, report=report_html, error=error)
